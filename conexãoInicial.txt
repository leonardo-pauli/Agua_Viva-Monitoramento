#include <WiFi.h>
#include <WebSocketsServer.h>

// ---------------- CONFIGURAÇÕES DA REDE LOCAL ----------------
const char* ssid = "Bioglow_Boia";      // Nome da rede Wi-Fi que o ESP32 vai criar
const char* password = "bioglowadmin";  // Senha (mínimo de 8 caracteres)
// -------------------------------------------------------------

// Inicializa o servidor WebSocket na porta 81
WebSocketsServer webSocket = WebSocketsServer(81);

// Variáveis de simulação
int bateriaSimulada = 100;
unsigned long tempoUltimoEnvio = 0;
const int intervaloEnvio = 2000; // Envia a cada 2 segundos

// Função que gere os eventos de conexão do aplicativo
void webSocketEvent(uint8_t num, WStype_t type, uint8_t * payload, size_t length) {
  switch(type) {
    case WStype_DISCONNECTED:
      Serial.printf("[%u] App Receptor desconectado!\n", num);
      break;
    case WStype_CONNECTED: {
      IPAddress ip = webSocket.remoteIP(num);
      Serial.printf("[%u] App Receptor conectado! IP: %d.%d.%d.%d\n", num, ip[0], ip[1], ip[2], ip[3]);
      break;
    }
  }
}

void setup() {
  Serial.begin(115200);
  Serial.println();

  // 1. Inicia o ESP32 como Access Point (Roteador)
  Serial.println("Configurando rede Wi-Fi Bioglow...");
  WiFi.softAP(ssid, password);

  // O IP padrão do ESP32 no modo AP é sempre 192.168.4.1
  IPAddress IP = WiFi.softAPIP();
  Serial.print("Rede criada com sucesso! IP da Boia: ");
  Serial.println(IP);

  // 2. Inicia o servidor WebSocket
  webSocket.begin();
  webSocket.onEvent(webSocketEvent);
  Serial.println("Servidor de dados iniciado na porta 81.");
}

void loop() {
  // Mantém a conexão do WebSocket viva e escutando o app
  webSocket.loop();

  // Envia os dados de tempos em tempos
  if (millis() - tempoUltimoEnvio > intervaloEnvio) {
    tempoUltimoEnvio = millis();

    // --- GERAÇÃO DOS DADOS SIMULADOS ---
    double frequenciaSimulada = random(80, 220) / 10.0; 
    
    if (bateriaSimulada > 1) {
      bateriaSimulada--;
    } else {
      bateriaSimulada = 100; 
    }

    // Como o ESP32 é o próprio roteador agora, o sinal Wi-Fi (RSSI) da placa não faz muito sentido.
    // Vamos simular uma variação de sinal para o seu app continuar bonito visualmente.
    int sinalSimulado = random(-75, -45); 

    // --- MONTAGEM DO JSON MANUAL ---
    String json = "{\"frequencia\": " + String(frequenciaSimulada) + 
                  ", \"bateria\": " + String(bateriaSimulada) + 
                  ", \"sinal\": " + String(sinalSimulado) + "}";

    // Faz o broadcast (envia) para todos os celulares conectados
    webSocket.broadcastTXT(json);

    Serial.print("Enviado para o App: ");
    Serial.println(json);
  }
}
